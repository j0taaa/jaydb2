from pathlib import Path
import hashlib,json,os,random,resource,struct,subprocess,sys

ROOT=Path(__file__).resolve().parents[3]
OUT=Path(__file__).resolve().parent
sys.path.insert(0,str(ROOT))
from tools.client import encode,decode
LO,HI=-(1<<63),(1<<63)-1
rng=random.Random(563221)

def native(name):
    source=OUT/(name+'.c');exe=OUT/name
    subprocess.run(['node','--stack_size=65536',str(ROOT/'tests/native/emit_fixture.mjs'),
      str(ROOT/'.tools/bend-source-2.0.39/bend2'),str(OUT/(name+'.bend')),str(source)],check=True)
    emitted=hashlib.sha256(source.read_bytes()).hexdigest()
    subprocess.run(['/usr/bin/clang-19','-std=gnu11','-O2','-w','--no-default-config',str(source),'-o',str(exe),'-lm','-pthread'],check=True)
    assert hashlib.sha256(source.read_bytes()).hexdigest()==emitted
    return exe,emitted

def run(exe,body):
    return subprocess.run([str(exe),'+RTS','--threads','1','--gpu','off','-RTS'],input=body,capture_output=True,
      timeout=90,check=True,env=dict(os.environ,BENDRTS='')).stdout

def checked(n):
    return b'\1'+struct.pack('<q',n) if LO<=n<=HI else b'\0'

def expected(op,a,b):
    if op==0:return checked(a+b)
    if op==1:return checked(a-b)
    if op==2:return checked(a*b)
    if op==4:return bytes([a<b])
    if b==0 or (a==LO and b==-1):return b'\0'
    q=(abs(a)//abs(b))*(-1 if (a<0)!=(b<0) else 1)
    return b'\1'+struct.pack('<qq',q,a-q*b)

def reference_key(xs):
    return ''.join('p'+''.join('c'+c for c in x)+'e' for x in xs)+'z'

def reference_decode(s):
    out=[];i=0
    while i<len(s):
      if s[i:]=='z':return out
      if s[i]!='p':return None
      i+=1;word=''
      while i<len(s) and s[i]=='c':
        if i+1==len(s):return None
        word+=s[i+1];i+=2
      if i==len(s) or s[i]!='e':return None
      out.append(word);i+=1
    return None

def main():
    _,hard=resource.getrlimit(resource.RLIMIT_STACK);resource.setrlimit(resource.RLIMIT_STACK,(hard,hard))
    arithmetic,ah=native('ArithmeticRuntime');keys,kh=native('KeysRuntime')
    pairs=set()
    divisors={0,1,-1,2,-2,3,-3,7,-7,HI,LO}
    for bit in range(1,63):
      for shift in (-1,0,1):
        d=(1<<bit)+shift
        if d<=HI:divisors.update((d,-d))
    for d in divisors:
      for a in [LO,LO+1,-1,0,1,HI-1,HI]:pairs.add((a,d))
      if d:
       for bound in (HI//abs(d),abs(LO)//abs(d)):
        for offset in (-2,-1,0,1,2):
         for sign in (-1,1):
          a=sign*(bound+offset)
          if LO<=a<=HI:pairs.update(((a,d),(d,a)))
    for _ in range(5000):
      a=rng.randrange(LO,HI+1);b=rng.choice([rng.randrange(LO,HI+1),rng.randrange(-65536,65537)])
      pairs.update(((a,b),(b,a)))
    cases=[(op,a,b) for a,b in pairs for op in range(5)];rng.shuffle(cases)
    for base in range(0,len(cases),700):
      chunk=cases[base:base+700]
      body=b''.join(bytes([op])+struct.pack('<qq',a,b) for op,a,b in chunk)
      exp=b''.join(expected(*c) for c in chunk)
      actual=run(arithmetic,body)
      assert actual==exp,(base,len(actual),len(exp),next((i for i,(a,b) in enumerate(zip(actual,exp)) if a!=b),None))
    alphabet=['','c','e','p','z',':','\x00','é','e\u0301','🫖','\U0010ffff']
    tuples=[[],[''],['','',''],alphabet]
    tuples += [[rng.choice(alphabet) for _ in range(rng.randrange(7))] for _ in range(500)]
    valid=[reference_key(t) for t in tuples]
    bad=['','p','pc','pe','zz','pz','pec','pecz','pezx']
    for s in valid:
      at=rng.randrange(len(s)+1)
      bad.extend([s[:at]+rng.choice(['p','c','e','z','\x00','🫖'])+s[at:],s[:at]+s[at+1:]])
    inputs=tuples+valid+bad
    exp=[reference_key(t) for t in tuples]+[reference_decode(s) for s in valid+bad]
    actual=decode(run(keys,encode(inputs)))
    assert actual==exp,next((i for i,(a,b) in enumerate(zip(actual,exp)) if a!=b),None)
    unique={}
    for t,key in zip(tuples,valid):
      if key in unique:assert unique[key]==t
      unique[key]=t
    report={'arithmetic_cases':len(cases),'distinct_operand_pairs':len(pairs),'key_cases':len(inputs),
      'arithmetic_emitted_sha256':ah,'key_emitted_sha256':kh,'status':'passed','controller_modified':False}
    (OUT/'result.json').write_text(json.dumps(report,indent=2)+'\n');print(json.dumps(report),flush=True)

if __name__=='__main__':main()
